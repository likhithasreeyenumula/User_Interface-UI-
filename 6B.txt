import 'package:flutter/material.dart';

void main() => runApp(const UnifiedBrandingApp());

class UnifiedBrandingApp extends StatelessWidget {
  const UnifiedBrandingApp({Key? key}) : super(key: key);
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      // Global Central Theme Schema Declaration Configuration
      theme: ThemeData(
        brightness: Brightness.light,
        scaffoldBackgroundColor: const Color(0xFFF7FAFC),
        primaryColor: const Color(0xFF2B6CB0),
        textTheme: const TextTheme(
          displayLarge: TextStyle(fontSize: 22, fontWeight: FontWeight.bold, color: Color(0xFF2D3748)),
          bodyLarge: TextStyle(fontSize: 14, color: Color(0xFF4A5568)),
        ),
      ),
      home: const BrandedThemeShowcaseView(),

  }
}

      appBar: AppBar(title: const Text('Design System Matrix')),
      body: Padding(
        padding: const EdgeInsets.all(24.0),

              'This text automatically inherits unified typography properties from the global application configuration rules.',
              style: centralBodyStyle,
            ),
          ],
        ),
      ),
    );
  }
}
