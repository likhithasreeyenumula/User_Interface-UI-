import 'package:flutter/material.dart';
void main() => runApp(const ResponsiveAdaptationApp());
class ResponsiveAdaptationApp extends StatelessWidget {
  const ResponsiveAdaptationApp({Key? key}) : super(key: key);
  @override
  Widget build(BuildContext context) {
    return const MaterialApp(
      debugShowCheckedModeBanner: false,
      home: ResponsiveDashboardWorkspace(),
    );
  }
}

class ResponsiveDashboardWorkspace extends StatelessWidget {
  const ResponsiveDashboardWorkspace({Key? key}) : super(key: key);
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Adaptive Scaling Console')),
      body: LayoutBuilder(
        builder: (BuildContext context, BoxConstraints structuralConstraints) {
          // Conditional toggle path check based on layout width bounds
          if (structuralConstraints.maxWidth > 700) {
            return _buildWideDesktopGrid();
          } else {
            return _buildMobileLinearStack();
          }
        },
      ),
    );
  }

  Widget _buildMobileLinearStack() {
    return ListView(
      padding: const EdgeInsets.all(16),
      children: [
        Container(
            height: 120,
            color: Colors.teal,
            child: const Center(child: Text('Mobile Workspace Node'))),
        const SizedBox(height: 12),
        Container(
            height: 120,
            color: Colors.teal,
            child: const Center(child: Text('Data Stream Monitor'))),
      ],
    );
  }

  Widget _buildWideDesktopGrid() {
    return Row(
      children: [
        Expanded(
          flex: 2,
          child: Container(
              color: Colors.blueGrey,
              child: const Center(child: Text('Admin Sidebar Workspace'))),
        ),
        Expanded(
          flex: 5,
          child: GridView.count(
            crossAxisCount: 2,
            padding: const EdgeInsets.all(16),
            children: [
              Card(
                  color: Colors.teal[100],
                  child: const Center(child: Text('Data Matrix Node Alpha'))),
              Card(
                  color: Colors.teal[100],
                  child: const Center(child: Text('Data Matrix Node Beta'))),
            ],
          ),
        )
      ],
    );
  }
}
