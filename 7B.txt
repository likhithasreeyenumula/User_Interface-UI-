import 'package:flutter/material.dart';

void main() {
  runApp(const MaterialApp(home: ValidatedFormWorkspace()));
}

class ValidatedFormWorkspace extends StatefulWidget {
  const ValidatedFormWorkspace({Key? key}) : super(key: key);
  @override
  State<ValidatedFormWorkspace> createState() => _ValidatedFormWorkspaceState();
}

class _ValidatedFormWorkspaceState extends State<ValidatedFormWorkspace> {
  // Global form key state handle mapping directly to internal validation layers
  final GlobalKey<FormState> _formValidationStateKey = GlobalKey<FormState>();

  void _commitDataPackage() {
    // Intercept and evaluate active field validation criteria
    if (_formValidationStateKey.currentState!.validate()) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Form Validation Confirmed. Processing data file...'),
          backgroundColor: Colors.green,
        ),
      );
    } else {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Submission Blocked: Input anomalies found.'),
          backgroundColor: Colors.red,
        ),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Validation Infrastructure Engine')),
      body: Padding(
        padding: const EdgeInsets.all(20.0),
        child: Form(
          key: _formValidationStateKey,
          child: Column(
            children: [
              TextFormField(
                decoration: const InputDecoration(labelText: 'Profile Identity Name'),
                validator: (String? value) {
                  if (value == null || value.trim().isEmpty) {
                    return 'Input Violation: Identity string cannot be empty.';
                  }
                  if (value.length < 3) {
                    return 'Input Violation: Minimum length requires 3 characters.';
                  }
                  return null;
                },
              ),
              const SizedBox(height: 25),
              ElevatedButton(
                onPressed: _commitDataPackage,
                child: const Text('Commit Account Data Package'),
              )
            ],
          ),
        ),
      ),
    );
  }
}
